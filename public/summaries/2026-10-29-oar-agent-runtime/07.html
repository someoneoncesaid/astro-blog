PATCH · YAML
- insert: - name: '@deepseek-ai/dsh-experimental-subagent-oar' config: runtime: codex providerName: oar-codex launchArgs: ['-c', 'sandbox_mode="workspace-write"'] - name: '@deepseek-ai/dsh-experimental-tool-subagent-oar' config: providers: codex: oar-codex
